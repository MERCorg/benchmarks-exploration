sort D = struct d1 | d2 | d3 | d4;
     Bit = struct e0 | e1;
     Frame = struct frame(dat: D, bit1: Bit, bit2: Bit);
     Status = struct read?eq_read | choice?eq_choice | del?eq_del;
     DBuf = List(D);

map  inv: Bit -> Bit;
     append: DBuf # D -> DBuf;
     first: DBuf -> D;
     next: DBuf -> DBuf;

var  l: DBuf;
     d,e: D;
eqn  inv(e0)  =  e1;
     inv(e1)  =  e0;
     append([], d)  =  [d];
     append(e |> l, d)  =  e |> append(l, d);
     first(d |> l)  =  d;
     next(d |> l)  =  l;

act  ra,sb,rc,sd: D;
     rcl,scl,i_del,i_lost,ccl;
     r,s,c,s2,r2,c2,s4,r4,c4: Frame;

proc P(l_Buffer1,l_Buffer2: List(D)) =
       (!(l_Buffer1 == [])) ->
         sb(first(l_Buffer1)) .
         P(l_Buffer1 = next(l_Buffer1))
     + sum d_Buffer1: D.
         (!(#l_Buffer1 == 2)) ->
         ra(d_Buffer1) .
         P(l_Buffer1 = append(l_Buffer1, d_Buffer1))
     + (!(l_Buffer2 == [])) ->
         sd(first(l_Buffer2)) .
         P(l_Buffer2 = next(l_Buffer2))
     + sum d_Buffer2: D.
         (!(#l_Buffer2 == 2)) ->
         rc(d_Buffer2) .
         P(l_Buffer2 = append(l_Buffer2, d_Buffer2))
     + delta;

init P([], []);
