sort D = struct d1 | d2 | d3;
     DBuf = List(D);
     BBuf = List(Bool);

map  N: Pos;
     empty: BBuf;
     insert: D # Nat # DBuf -> DBuf;
     insert: Bool # Nat # BBuf -> BBuf;
     nextempty_mod: Nat # BBuf # Nat # Pos -> Nat;
     constDbuf: Nat -> DBuf;
     constBbuf: Nat -> BBuf;
     q1,q2: DBuf;
     append: DBuf # D -> DBuf;
     first: DBuf -> D;
     next: DBuf -> DBuf;

var  d,d',e: D;
     i,j,m: Nat;
     q,l: DBuf;
     c,c': Bool;
     n': Pos;
     b: BBuf;
eqn  N  =  2;
     (i == 0)  ->  constDbuf(i)  =  [];
     (i == 0)  ->  constBbuf(i)  =  [];
     (i > 0)  ->  constDbuf(i)  =  d1 |> constDbuf(Int2Nat(i - 1));
     (i > 0)  ->  constBbuf(i)  =  false |> constBbuf(Int2Nat(i - 1));
     q1  =  constDbuf(N);
     q2  =  q1;
     empty  =  constBbuf(N);
     (i == 0)  ->  insert(d, i, q)  =  d |> tail(q);
     (i > 0)  ->  insert(d, i, d' |> q)  =  d' |> insert(d, Int2Nat(i - 1), q);
     (i == 0)  ->  insert(c, i, b)  =  c |> tail(b);
     (i > 0)  ->  insert(c, i, c' |> b)  =  c' |> insert(c, Int2Nat(i - 1), b);
     (b . (i mod n') && m > 0)  ->  nextempty_mod(i, b, m, n')  =  nextempty_mod((i + 1) mod 2 * n', b, Int2Nat(m - 1), n');
     (!(b . (i mod n') && m > 0))  ->  nextempty_mod(i, b, m, n')  =  i mod 2 * n';
     append([], d)  =  [d];
     append(e |> l, d)  =  e |> append(l, d);
     first(d |> l)  =  d;
     next(d |> l)  =  l;

act  r1,s4: D;
     s2,r2,c2,s3,r3,c3: D # Nat;
     s5,r5,c5,s6,r6,c6: Nat;
     i;

proc P(l_Buffer: List(D), b_Buffer: Bool) =
       (!(l_Buffer == []) && (b_Buffer || !b_Buffer)) ->
         s4(first(l_Buffer)) .
         P(l_Buffer = next(l_Buffer), b_Buffer = true)
     + sum d_Buffer: D.
         (!(#l_Buffer == 4)) ->
         r1(d_Buffer) .
         P(l_Buffer = append(l_Buffer, d_Buffer), b_Buffer = false)
     + delta;

init P([], true);
